PATH-G1 wrist D405 ablation launch commands (v11)
===================================================

Each experiment below is self-contained. Use exactly one experiment at a time,
after stopping the previous SAM3 worker, D405 gateway, and PATH bridge.

Terminal A and B run on Thor. Terminal C runs on G1. Paste each whole terminal
block into its corresponding terminal. The two Thor terminals must use the same
BELIEF_CONFIG for one experiment.

The commands assume that Thor already has the required image and TensorRT engine:
  /models/sam3_trt/sam3_fp16_4out.engine

The gateway uses --auto-devices to detect the two attached D405 cameras. Do not
use the RTX-4060-only run_remote_sam3_wrist_d405_v11.sh on Thor.


0. PATH baseline (belief position replacement + orientation belief)
--------------------------------------------------------------------

Terminal A -- Thor SAM3 worker

cd ~/PATH-G1/perception
export WIFI_LCM_URL='udpm://239.255.77.67:7667?ttl=1'
export BELIEF_CONFIG='config/pipeline.yaml'

./tools/run_thor.sh python -m remote_offload.remote_sam3_wrist_d405_worker_v11 \
  --wifi-lcm-url "$WIFI_LCM_URL" \
  --backend tensorrt \
  --trt-engine /models/sam3_trt/sam3_fp16_4out.engine \
  --concepts object \
  --extrinsics config/wrist_d405_extrinsics_v11_translation_calibrated.yaml \
  --belief-config "$BELIEF_CONFIG" \
  --pairing-max-residual-ms 15 \
  --pairing-max-error-m 0.008

Terminal B -- Thor D405 capture and belief gateway

cd ~/PATH-G1/perception
export G1_LCM_URL='udpm://239.255.76.67:7667?ttl=1'
export WIFI_LCM_URL='udpm://239.255.77.67:7667?ttl=1'
export BELIEF_CONFIG='config/pipeline.yaml'

./tools/run_thor_wrist_d405_gateway_v11.sh \
  --auto-devices \
  --g1-lcm-url "$G1_LCM_URL" \
  --wifi-lcm-url "$WIFI_LCM_URL" \
  --config config/wrist_d405_v11.yaml \
  --belief-config "$BELIEF_CONFIG" \
  --max-hz 8 \
  --camera-recovery-initial-delay-ms 300 \
  --camera-recovery-max-delay-ms 2000 \
  --belief-inference \
  --belief-input-source operator_command \
  --belief-command-max-delta-ms 25 \
  --belief-command-divergence-max-m 0.20 \
  --ee-target-publishing \
  --log-level INFO

Terminal C -- G1 PATH state-machine bridge

cd ~/PATH-G1
export PATH_LCM_URL='udpm://239.255.76.67:7667?ttl=1'
export PATH_WRIST_VISUAL_LCM_URL='udpm://239.255.77.67:7667?ttl=1'
export PATH_SHARED_POSITION_MODE='replace'
unset PATH_SHARED_POSITION_ALPHA
unset PATH_SHARED_POSITION_ALPHA_SOURCE
unset PATH_SHARED_POSITION_POSTERIOR_ENTER_CONFIDENCE
unset PATH_SHARED_POSITION_POSTERIOR_EXIT_CONFIDENCE
unset PATH_SHARED_POSITION_POSTERIOR_EMA_ALPHA
unset PATH_SHARED_POSITION_POSTERIOR_MAX_RATE_PER_S

./path_deploy/scripts/run_path_direct_lcm.sh
